test('ambiguous entry needs one choice before selection and zero afterward without changing identity/status',()=>{
 const e=entry('amb'),h=harness([e]),selected=e.matches[1];assert.equal(h.count(),1);assert.equal(h.c.selected,null);h.choose(1);assert.equal(h.count(),0);assert.equal(e.status,'ambiguous');assert.equal(e.sourceChoiceConfirmed,true);assert.equal(e.selectedRecord,selected);assert.equal(e.matches.length,2);assert.equal(h.lookups,0);
});
test('singleton needs an explicit choice even when a selected pointer exists; choosing clears its count',()=>{
 const e=entry('single','source-match',1),h=harness([e]);assert.equal(h.count(),1);assert.equal(e.selectedRecord,null);h.c.selected=e.matches[0];h.c.render();assert.equal(h.count(),1);assert.equal(e.sourceChoiceConfirmed,false);h.choose();assert.equal(h.count(),0);assert.equal(e.selectedRecord,e.matches[0]);
});
test('unavailable/error/pending/no-candidate entries do not ask for an impossible choice; partial loaded records do',()=>{
 const entries=[entry('unavailable','unavailable',0),entry('error','error',0),entry('pending','pending',0),entry('empty','source-match',0)],h=harness(entries);assert.equal(h.count(),0);
 const p=entry('partial','partial',1);entries.push(p);h.c.current=p;h.c.selected=null;assert.equal(h.count(),1);h.choose();assert.equal(h.count(),0);
 const inFlight=entry('inflight','source-match',1);inFlight.lookupPending=true;entries.push(inFlight);assert.equal(h.count(),0);
});
test('confirmation is insufficient when its saved record is missing or outside the same entry matches',()=>{

test('actual Undo restores preceding confirmed or unresolved state and the rendered count follows the remaining list',()=>{
